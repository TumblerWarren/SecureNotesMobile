cmake_minimum_required(VERSION 3.22.1)
project(securenote LANGUAGES C)
set(SODIUM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.sodium/${ANDROID_ABI}")
if(NOT EXISTS "${SODIUM_ROOT}/lib/libsodium.a")
    message(FATAL_ERROR "Run python3 tools/build_sodium.py --ndk <NDK path> first")
endif()
add_library(sodium STATIC IMPORTED)
set_target_properties(sodium PROPERTIES IMPORTED_LOCATION "${SODIUM_ROOT}/lib/libsodium.a")
add_library(vault_kdf SHARED sodium_jni.c legacy_sqlite.c native_io.c)
target_include_directories(vault_kdf PRIVATE "${SODIUM_ROOT}/include")
target_link_libraries(vault_kdf sodium dl)
target_link_options(vault_kdf PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,relro,-z,now")
